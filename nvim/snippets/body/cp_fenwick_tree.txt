struct fenwick_tree {
    int n;
    vector<int> BIT;

    fenwick_tree(int n) : n(n), BIT(n) {};

    void build(const vector<int>& a) {
        for (int i = 0; i < n; i++) {
            BIT[i] += a[i];
            const int r = i | (i + 1);
            if (r < n) BIT[r] += BIT[i];
        }
    }

    void add(int idx, int val) {
        for (int i = idx; i < n; i = i | (i + 1)) {
            BIT[i] += val;
        }
    }

    int sum(int r) {
        int res = 0;
        for (int i = r; i >= 0; i = (i & (i + 1)) - 1) {
            res += BIT[i];
        }
        return res;
    }

    int sum(int l, int r) {
        return sum(r) - sum(l - 1);
    }
};
